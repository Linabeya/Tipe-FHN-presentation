\section{Analyse mathématique du système FHN}

\begin{frame}
\frametitle{Analyse mathématique du système FHN}
\framesubtitle{Modèle et portrait de phase}

\begin{columns}

\column{0.48\textwidth}

\alert{\textbf{Équations du modèle}}
\[
\begin{cases}
\dot v = v-\dfrac{v^3}{3}-w+I\\[4pt]
\dot w = \varepsilon(v+a-bw)
\end{cases}
\]

\medskip
\alert{\textbf{Isoclines (nullclines)}} 
\begin{itemize} 
  \item $\dot v=0$ donne $w = v - \dfrac{v^3}{3} + I$ 
  \item $\dot w=0$ donne $w = \dfrac{v+a}{b}$ 
\end{itemize} 
\column{0.5\textwidth} 
\centering 
\begin{figure} 
  \includegraphics[width=\linewidth]{Figures/FHN.png} 
  \caption{Portrait de phase du modèle de FitzHugh-Nagumo} 
\end{figure} 
\end{columns} 
\vspace{0.2cm} 
\centerline{L’intersection des isoclines donne le(s) point(s) fixe(s).} 
\end{frame}

\begin{frame} 
  \frametitle{Analyse mathématique du système FHN} 
  \framesubtitle{Point(s) fixe(s) et stabilité linéaire} \begin{itemize} \item \alert{Point(s) fixe(s) :} intersection des deux isoclines. \[ \begin{cases} w^* = v^* - \dfrac{v^{*3}}{3} + I \\[6pt] w^* = \dfrac{v^*+a}{b} \end{cases} \] \item Selon $I$, $a$ et $b$, on obtient 1 ou 3 points fixes. Nous considérons le cas usuel d’un point fixe unique. \end{itemize} \alert{Linéarisation autour de $(v^*,w^*)$} \[ \mathbf{J} = \begin{pmatrix} 1 - v^{*2} & -1 \\ \varepsilon & -\varepsilon b \end{pmatrix} \] \begin{itemize} \item Trace : $\operatorname{tr}(\mathbf{J}) = 1 - v^{*2} - \varepsilon b$ \item Déterminant : $\det(\mathbf{J}) = \varepsilon\bigl[1 + b(v^{*2}-1)\bigr]$ \end{itemize} \end{frame}

\begin{frame}
\frametitle{Analyse mathématique du système FHN}
\framesubtitle{Nature du point fixe}

\alert{\textbf{Polynôme caractéristique}}

\[
\lambda^2-\operatorname{tr}(J)\lambda+\det(J)=0
\]

\[
\lambda_{\pm}
=
\frac{\operatorname{tr}(J)\pm\sqrt{\Delta}}{2}
\]

\[
\Delta=
\operatorname{tr}(J)^2-4\det(J)
\]

\alert{\textbf{Classification}}

\begin{itemize}
\item $\operatorname{tr}(J)<0,\ \Delta>0$
      : nœud stable.

\item $\operatorname{tr}(J)<0,\ \Delta<0$
      : foyer stable.

\item $\operatorname{tr}(J)>0$
      : point fixe instable.

\item $\det(J)<0$
      : point selle.
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Analyse mathématique du système FHN}
\framesubtitle{Bifurcation de Hopf}

\alert{\textbf{Condition critique}}
\[
\operatorname{tr}(J)=0
\]
soit
\[
1-{v^*}^2=\varepsilon b
\]
Avec
\[
\det(J)>0,
\]
les valeurs propres deviennent
\[
\lambda_{\pm}
=
\pm i\sqrt{\det(J)}
\]

\bigskip

\alert{\textbf{Interprétation physique}}

\begin{itemize}
\item Lorsque \(I\), \(a\) ou \(b\) varie, le point fixe se déplace sur la cubique.

\item En franchissant la condition
\(
1-{v^*}^2=\varepsilon b
\),
la stabilité change.

\item Un cycle limite apparaît :
le système passe d'un état de repos à des oscillations auto-entretenues.
\end{itemize}
\end{frame}
